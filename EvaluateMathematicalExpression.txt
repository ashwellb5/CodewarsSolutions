def find_expression(exp):
    operators = ["+", "-", "*","/"] 
    mult_divide = False
    if "*" in exp or "/" in exp:
        mult_divide = True
    num1 = ""
    num2 = ""
    negate = ""
    operator = ""
    num1bool = False
    num2bool = False
    index = 0
    start_index = -1
    end_index = -1
    for x in exp:
        if x.isdigit() or x == ".":
            if num1 == "" or num1bool == True:
                if start_index == -1:
                    if negate == "-":
                        start_index = index - 1
                    else:
                        start_index = index
                num1 += negate + x
                num1bool = True
            elif num2 == "" or num2bool == True:
                num2 += negate + x
                num2bool = True
            negate = ""
        elif (x == "-" and negate == "") and ((operator != "" or num1 == "") and num2 == ""):
            negate = "-"
        elif x in operators and operator == "":
            if mult_divide and (x == "+" or x == "-"):
                num1 = ""
                num2 = ""
                start_index = -1
            else:
                operator = x
            negate = ""
            num1bool = False
            num2bool = False
        else:
            if (num1 != "" and num2 != "") and end_index == -1:
                end_index = index
            negate = ""
            num1bool = False
            num2bool = False
        index += 1
    if end_index == -1:
        end_index = index
    return [num1, num2, operator, start_index, end_index]

def simple_expression(exp):
    if exp[2] == "+":
        return float(exp[0]) + float(exp[1])
    elif exp[2] == "-":
        return float(exp[0]) - float(exp[1])
    elif exp[2] == "*":
        return float(exp[0]) * float(exp[1])
    elif exp[2] == "/":
        return float(exp[0]) / float(exp[1])
